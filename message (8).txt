local function safeRequire(instance)

    end

local function normalizeName(value)

    if type(value) == "string" then

        return value

    end

    if type(value) == "table" and type(value.Name) == "string" then

        return value.Name

    end

    if typeof(value) == "EnumItem" then

        return value.Name

    end

    return nil

end

local function normalizeType(value)

    local name = normalizeName(value)

    if name == "Wrapping" then

        return "Wrap"

    end

    return name

end

local function hasUnobtainableMarker(value)

    local valueType = typeof(value)

    if valueType == "string" or valueType == "EnumItem" then

        return tostring(value):lower():find("unobtainable", 1, true) ~= nil

    end

    if type(value) == "table" then

        for _, key in ipairs({"Name", "Value", "Type", "Rarity", "Status"}) do

            local nested = rawget(value, key)

            if nested ~= nil and nested ~= value and hasUnobtainableMarker(nested) then

                return true

            end

        end

    end

    return false

end

local function isUnobtainable(cosmetic)

    if type(cosmetic) ~= "table" then

        return false

    end

    if cosmetic.Unobtainable == true then

        return true

    end

    for _, key in ipairs({

        "Rarity",

        "RarityName",

        "Availability",

        "Obtainability",

        "Status",

        "Category",

    }) do

        if hasUnobtainableMarker(cosmetic[key]) then

            return true

        end

    end

    local tags = cosmetic.Tags

    if type(tags) == "table" then

        for key, value in pairs(tags) do

            if hasUnobtainableMarker(key) or hasUnobtainableMarker(value) then

                return true

            end

        end

    end

    return false

end

local AllowedCosmeticType = {}

for cosmeticName, cosmetic in pairs(CosmeticLibrary.Cosmetics or {}) do

    if type(cosmeticName) == "string"

        and type(cosmetic) == "table"

        and not isUnobtainable(cosmetic) then

        local cosmeticType = normalizeType(cosmetic.Type)

        if ALL_TYPES[cosmeticType] then

            AllowedCosmeticType[cosmeticName] = cosmeticType

        end

    end

end

local function isSupportedCosmetic(name, expectedType)

    if type(name) ~= "string" or name == "" then

        return false

    end

    local cosmeticType = AllowedCosmeticType[name]

    if not cosmeticType then

        return false

    end

    if expectedType then

        return cosmeticType == normalizeType(expectedType)

    end

    return true

end

    end

end

local function saveConfig()

    if type(writefile) ~= "function" then

        return false

    end

    local serializable = {

        version = 2,

        weapons = {},

        emotes = {},

        wrapInverted = {},

    }

    for weaponName, selections in pairs(State.weapons) do

        if type(weaponName) == "string" and type(selections) == "table" then

            local clean = {}

            for _, kind in ipairs({"Skin", "Wrap", "Charm", "Finisher"}) do

                local value = selections[kind]

                if type(value) == "string" and value ~= "" then

                    if value == NONE or isSupportedCosmetic(value, kind) then

                        clean[kind] = value

                    end

                end

            end

            if next(clean) ~= nil then

                serializable.weapons[weaponName] = clean

            end

        end

    end

    for slotKey, emoteName in pairs(State.emotes) do

        if type(emoteName) == "string" and isSupportedCosmetic(emoteName, "Emote") then

            serializable.emotes[tostring(slotKey)] = emoteName

        end

    end

    for weaponName, inverted in pairs(State.wrapInverted) do

        if type(weaponName) == "string" then

            serializable.wrapInverted[weaponName] = inverted == true

        end

    end

    ensureConfigFolder()

    local ok, encoded = pcall(HttpService.JSONEncode, HttpService, serializable)

    if not ok or type(encoded) ~= "string" then

        return false

    end

    local writeOk = pcall(writefile, CONFIG_FILE, encoded)

    if writeOk and type(queueSyncPublish) == "function" then

        queueSyncPublish()

    end

    return writeOk

end

local function loadConfig()

    if type(readfile) ~= "function" or type(isfile) ~= "function" or not isfile(CONFIG_FILE) then

        return false

    end

    local ok, raw = pcall(readfile, CONFIG_FILE)

    if not ok or type(raw) ~= "string" then

        return false

    end

    local decodeOk, decoded = pcall(HttpService.JSONDecode, HttpService, raw)

    if not decodeOk or type(decoded) ~= "table" then

        return false

    end

    State.weapons = {}

    State.emotes = {}

    State.wrapInverted = {}

    if type(decoded.weapons) == "table" then

        for weaponName, selections in pairs(decoded.weapons) do

            if type(weaponName) == "string" and type(selections) == "table" then

                local clean = {}

                for _, kind in ipairs({"Skin", "Wrap", "Charm", "Finisher"}) do

                    local value = selections[kind]

                    if type(value) == "string" then

                        if value == NONE or isSupportedCosmetic(value, kind) then

                            clean[kind] = value

                        end

                    end

                end

                if next(clean) ~= nil then

                    State.weapons[weaponName] = clean

                end

            end

        end

    end

    if type(decoded.emotes) == "table" then

        for slotKey, emoteName in pairs(decoded.emotes) do

            if type(emoteName) == "string" and isSupportedCosmetic(emoteName, "Emote") then

                State.emotes[tostring(slotKey)] = emoteName

            end

        end

    end

    if type(decoded.wrapInverted) == "table" then

        for weaponName, inverted in pairs(decoded.wrapInverted) do

            if type(weaponName) == "string" then

                State.wrapInverted[weaponName] = inverted == true

            end

        end

    end

    return true

end

local function resolveHttpRequest()

    if syncHttpRequest ~= nil then

        return syncHttpRequest or nil

    end

    local environment = _G

    if type(getgenv) == "function" then

        local ok, value = pcall(getgenv)

        if ok and type(value) == "table" then

            environment = value

        end

    end

    local candidates = {

        type(environment) == "table" and environment.request or nil,

        type(environment) == "table" and environment.http_request or nil,

        type(environment) == "table" and environment.httprequest or nil,

        type(environment) == "table" and environment.request_async or nil,

        type(request) == "function" and request or nil,

        type(http_request) == "function" and http_request or nil,

        type(httprequest) == "function" and httprequest or nil,

    }

    if type(syn) == "table" then

        candidates[#candidates + 1] = syn.request

    end

    if type(http) == "table" then

        candidates[#candidates + 1] = http.request

        candidates[#candidates + 1] = http.request_async

    end

    if type(fluxus) == "table" then

        candidates[#candidates + 1] = fluxus.request

    end

    if type(krnl) == "table" then

        candidates[#candidates + 1] = krnl.request

    end

    for _, candidate in ipairs(candidates) do

        if type(candidate) == "function" then

            syncHttpRequest = candidate

            return candidate

        end

    end

    syncHttpRequest = false

    return nil

end

local function writeSyncDiagnostic(data)

        return

    end

    ensureConfigFolder()

    if ok and type(encoded) == "string" then

            encoded

        )

    end

end

local function syncRequest(method, path, body)

    local requester = resolveHttpRequest()

    if type(requester) ~= "function" then

        writeSyncDiagnostic({

            ok = false,

            stage = "resolve_request",

            error = "no_http_request_function",

        })

        return nil

    end

    local options = {

        Url = SYNC_URL .. path,

        URL = SYNC_URL .. path,

        Method = method,

        Headers = {

            ["Content-Type"] = "application/json",

            ["Accept"] = "application/json",

        },

    }

    if body ~= nil then

        local ok, encoded =

            pcall(HttpService.JSONEncode, HttpService, body)

        if not ok then

            writeSyncDiagnostic({

                ok = false,

                stage = "encode",

                error = "json_encode_failed",

            })

            return nil

        end

        options.Body = encoded

    end

    local ok, response = pcall(requester, options)

    if not ok then

        writeSyncDiagnostic({

            ok = false,

            stage = "request",

            error = tostring(response),

            method = method,

            path = path,

        })

        return nil

    end

    if type(response) == "string" then

        local decodeOk, decoded =

            pcall(

                HttpService.JSONDecode,

                HttpService,

                response

            )

        if decodeOk and type(decoded) == "table" then

            writeSyncDiagnostic({

                ok = true,

                stage = "response",

                method = method,

                path = path,

                status = 200,

            })

            return decoded

        end

        writeSyncDiagnostic({

            ok = false,

            stage = "response",

            error = "string_response_not_json",

            method = method,

            path = path,

        })

        return nil

    end

    if type(response) ~= "table" then

        writeSyncDiagnostic({

            ok = false,

            stage = "response",

            error = "unsupported_response_type",

            responseType = typeof(response),

            method = method,

            path = path,

        })

        return nil

    end

    local statusCode =

        tonumber(response.StatusCode)

        or tonumber(response.Status)

        or tonumber(response.status_code)

        or tonumber(response.status)

    local responseBody =

        response.Body

        or response.body

        or response.ResponseBody

    if statusCode and

        (statusCode < 200 or statusCode >= 300) then

        writeSyncDiagnostic({

            ok = false,

            stage = "http_status",

            status = statusCode,

            body = tostring(responseBody or ""),

            method = method,

            path = path,

        })

        return nil

    end

    if type(responseBody) ~= "string"

        or responseBody == "" then

        writeSyncDiagnostic({

            ok = true,

            stage = "response",

            status = statusCode or 200,

            method = method,

            path = path,

            emptyBody = true,

        })

        return {}

    end

    local decodeOk, decoded =

        pcall(

            HttpService.JSONDecode,

            HttpService,

            responseBody

        )

    if not decodeOk or type(decoded) ~= "table" then

        writeSyncDiagnostic({

            ok = false,

            stage = "decode_response",

            status = statusCode or 200,

            body = tostring(responseBody),

            method = method,

            path = path,

        })

        return nil

    end

    writeSyncDiagnostic({

        ok = true,

        stage = "response",

        status = statusCode or 200,

        method = method,

        path = path,

    })

    return decoded

end

local function buildSyncWeapons()

    local weapons = {}

    for weaponName, selections in pairs(State.weapons) do

        if type(weaponName) == "string"

            and type(selections) == "table" then

            local clean = {}

            local syncKinds = SHARE_WRAPS

                and {"Skin", "Wrap", "Charm", "Finisher"}

                or {"Skin", "Charm", "Finisher"}

            for _, kind in ipairs(syncKinds) do

                local cosmeticName = selections[kind]

                if type(cosmeticName) == "string"

                    and cosmeticName ~= NONE

                    and isSupportedCosmetic(

                        cosmeticName,

                        kind

                    ) then

                    clean[kind] = cosmeticName

                end

            end

            if SHARE_WRAPS and clean.Wrap ~= nil then

                clean.WrapInverted =

                    State.wrapInverted[weaponName] == true

            end

            if next(clean) ~= nil then

                weapons[weaponName] = clean

            end

        end

    end

    return weapons

end

local function refreshSharedCosmetics()

    if game.JobId == "" then

        return false

    end

    local response = syncRequest(

        "GET",

        "/v1/cosmetics?jobId="

            .. HttpService:UrlEncode(game.JobId)

    )

    if type(response) ~= "table"

        or response.ok ~= true

        or type(response.players) ~= "table" then

        return false

    end

    local nextShared = {}

    for userIdString, entry in pairs(response.players) do

        local userId = tonumber(userIdString)

        if userId

            and userId ~= player.UserId

            and type(entry) == "table"

            and type(entry.weapons) == "table" then

            local remotePlayer =

                Players:GetPlayerByUserId(userId)

            if remotePlayer then

                local clean =

                    sanitizeSharedWeapons(entry.weapons)

                if next(clean) ~= nil then

                    nextShared[userId] = clean

                end

            end

        end

    end

    SharedCosmetics = nextShared

    return true

end

    end)

end

local function setSelection(weaponName, kind, cosmeticName)

    kind = normalizeType(kind)

    if type(weaponName) ~= "string" or weaponName == "" or not WEAPON_TYPES[kind] then

        return false

    end

    local normalized = cosmeticName

    if normalized == "NONE_COSMETIC" or normalized == "" or normalized == nil then

        normalized = NONE

    elseif normalized == "RANDOM_COSMETIC" then

        normalized = RANDOM

    end

    if normalized ~= NONE and normalized ~= RANDOM and not isSupportedCosmetic(normalized, kind) then

        return false

    end

    State.weapons[weaponName] = State.weapons[weaponName] or {}

    if normalized == RANDOM then

        State.weapons[weaponName][kind] = nil

    else

        State.weapons[weaponName][kind] = normalized

    end

    if next(State.weapons[weaponName]) == nil then

        State.weapons[weaponName] = nil

    end

    saveConfig()

    return true

end

local CharmViewPayloadCache = {}

local function buildCharmViewPayload(charmName)

    if type(charmName) ~= "string" or not isSupportedCosmetic(charmName, "Charm") then

        return nil

    end

    local cached = CharmViewPayloadCache[charmName]

    if cached then

        return cached

    end

    local payload = {Name = charmName}

    local source = CosmeticLibrary.Cosmetics[charmName]

    if type(source) == "table" and source.Metadata ~= nil then

        payload.Metadata = source.Metadata

    end

    CharmViewPayloadCache[charmName] = payload

    return payload

end

local function buildCosmeticData(existingData, cosmeticName, cosmeticType)

    if cosmeticName == NONE then

        return nil

    end

    if not isSupportedCosmetic(cosmeticName, cosmeticType) then

        return existingData

    end

    local source = CosmeticLibrary.Cosmetics[cosmeticName]

    local data = {}

    if type(source) == "table" then

        for key, value in pairs(source) do

            data[key] = value

        end

    elseif type(existingData) == "table" then

        for key, value in pairs(existingData) do

            data[key] = value

        end

    end

    data.Name = cosmeticName

    data.Type = data.Type or cosmeticType

    data.Seed = data.Seed or math.random(1, 1000000)

    if EnumLibrary and type(EnumLibrary.ToEnum) == "function" then

        local ok, enumValue = pcall(EnumLibrary.ToEnum, EnumLibrary, cosmeticName)

        if ok and enumValue ~= nil then

            data.Enum = enumValue

            data.ObjectID = enumValue

        end

    end

    if data.Inverted == nil then

        data.Inverted = false

    end

    return data

end

local function applySelectionToWeaponData(weaponName, weaponData)

    if type(weaponData) ~= "table" then

        return weaponData

    end

    local selections = State.weapons[weaponName]

    if type(selections) ~= "table" then

        return weaponData

    end

    for _, kind in ipairs({"Skin", "Wrap", "Charm", "Finisher"}) do

        local selectedName = selections[kind]

        if selectedName ~= nil then

            if selectedName == NONE then

                weaponData[kind] = nil

            else

                local nextData = buildCosmeticData(weaponData[kind], selectedName, kind)

                if kind == "Wrap" and type(nextData) == "table" then

                    nextData.Inverted = State.wrapInverted[weaponName] == true

                end

                weaponData[kind] = nextData

            end

        end

    end

    return weaponData

end

local function applyWeaponInventory()

    local inventory = getRawData("WeaponInventory")

    if type(inventory) ~= "table" then

        return false

    end

    local changed = false

    for _, weaponData in pairs(inventory) do

        if type(weaponData) == "table" and type(weaponData.Name) == "string" then

            if State.weapons[weaponData.Name] then

                applySelectionToWeaponData(weaponData.Name, weaponData)

                changed = true

            end

        end

    end

    local currentData = DataController.CurrentData

    if changed and currentData and type(currentData.Replicate) == "function" then

        pcall(currentData.Replicate, currentData, "WeaponInventory")

    end

    return changed

end

DataController.Get = function(self, key)

    local data = originalGet(self, key)

    if key == "CosmeticInventory" then

        return buildUnlockedInventory(data)

    end

    if key == "WeaponInventory" and type(data) == "table" then

        for _, weaponData in pairs(data) do

            if type(weaponData) == "table" and type(weaponData.Name) == "string" then

                applySelectionToWeaponData(weaponData.Name, weaponData)

            end

        end

        return data

    end

    return data

end

local originalGetWeaponData = DataController.GetWeaponData

DataController.GetWeaponData = function(self, weaponName)

    local data = originalGetWeaponData(self, weaponName)

    if type(data) ~= "table" then

        return data

    end

    local merged = table.clone(data)

    merged.Name = weaponName

    return applySelectionToWeaponData(weaponName, merged)

end

local function captureNativeSelection(invertedOverride)

    local weaponName = normalizeName(readEquipmentStateValue("SelectedWeapon"))

    local cosmeticType = normalizeType(readEquipmentStateValue("CustomizingType"))

    local cosmeticName = normalizeName(readEquipmentStateValue("SelectedCosmetic"))

    if type(weaponName) ~= "string" or weaponName == "" or not WEAPON_TYPES[cosmeticType] then

        return false

    end

    if type(cosmeticName) ~= "string" or cosmeticName == "" or cosmeticName == "NONE_COSMETIC" then

        cosmeticName = NONE

    elseif cosmeticName == "RANDOM_COSMETIC" then

        cosmeticName = RANDOM

    end

    if cosmeticName ~= NONE and cosmeticName ~= RANDOM then

        if not isSupportedCosmetic(cosmeticName, cosmeticType) then

            return false

        end

    end

    if cosmeticType == "Wrap" then

        local inverted = invertedOverride

        if inverted == nil then

            local nativeInverted = readEquipmentStateValue("CosmeticInverted")

            inverted = nativeInverted == true

        end

        State.wrapInverted[weaponName] = inverted == true

    end

    local changed = setSelection(weaponName, cosmeticType, cosmeticName)

    if changed then

        applyWeaponInventory()

        triggerDataChanged("WeaponInventory")

        triggerDataChanged("WeaponData")

    end

    return changed

end

local function ensureNativeCosmeticSlotBinding()

    if nativeSlotBound then

    end

    local cosmeticSlot = resolveCosmeticSlot()

    local constructor = type(cosmeticSlot) == "table" and rawget(cosmeticSlot, "new") or nil

    if type(constructor) ~= "function"

        or type(debug.getupvalue) ~= "function"

        or type(debug.setupvalue) ~= "function" then

        return false

    end

    local prototype = debug.getupvalue(constructor, 1)

    if type(prototype) ~= "table" then

        return false

    end

    local proxy = {}

    proxy.__index = function(instance, key)

        if key ~= "Frame" then

            return rawget(prototype, key)

        end

        if type(debug.info) == "function" and type(debug.getupvalues) == "function" then

            local source, caller = debug.info(2, "sf")

            local upvalues = type(caller) == "function" and debug.getupvalues(caller) or nil

            if type(source) == "string"

                and source:match("Cosmetics$")

                and type(upvalues) == "table"

                and type(upvalues[9]) == "boolean" then

                pcall(debug.setupvalue, caller, 9, false)

            end

        end

        return rawget(instance, SLOT_FRAME_SENTINEL)

    end

    proxy.__newindex = function(instance, key, value)

        if key ~= "Frame" then

            rawset(instance, key, value)

            return

        end

        local source = type(debug.info) == "function" and debug.info(3, "s") or nil

        if type(source) == "string"

            and source:match("Cosmetics$")

            and rawget(instance, "IsLocked") ~= true

            and value ~= nil then

            local button = value:FindFirstChild("Button")

            if button ~= nil then

                button.MouseButton1Click:Connect(function()

                    task.defer(captureNativeSelection)

                end)

            end

        end

        rawset(instance, SLOT_FRAME_SENTINEL, value)

    end

    local ok = pcall(debug.setupvalue, constructor, 1, proxy)

    if not ok then

        return false

    end

    nativeSlotBound = true

    return true

end

local function ensureNativeInvertedState()

    if equipmentStateHooked then

        return true

    end

    local equipment = resolveEquipment()

    local equipmentState = type(equipment) == "table" and rawget(equipment, "EquipmentState") or nil

    local originalMetatable = type(equipmentState) == "table" and getmetatable(equipmentState) or nil

    if type(equipmentState) ~= "table" or type(originalMetatable) ~= "table" then

        return false

    end

    local originalIndex = rawget(originalMetatable, "__index")

    local originalNewIndex = rawget(originalMetatable, "__newindex")

    local nextMetatable = {}

    for key, value in pairs(originalMetatable) do

        nextMetatable[key] = value

    end

    nextMetatable.__index = function(target, key)

        if key == "CosmeticInverted" then

            local weaponName = normalizeName(rawget(target, "SelectedWeapon"))

            if type(weaponName) == "string" and State.wrapInverted[weaponName] ~= nil then

                return State.wrapInverted[weaponName] == true

            end

        end

        if type(originalIndex) == "table" then

            return originalIndex[key]

        elseif type(originalIndex) == "function" then

            return originalIndex(target, key)

        end

        return rawget(target, key)

    end

    nextMetatable.__newindex = function(target, key, value)

        if key == "CosmeticInverted" and value ~= nil then

            local weaponName = normalizeName(rawget(target, "SelectedWeapon"))

            if type(weaponName) == "string" and weaponName ~= "" then

                State.wrapInverted[weaponName] = value == true

                saveConfig()

                task.defer(function()

                    captureNativeSelection(value == true)

                end)

            end

            rawset(target, key, value)

            return

        end

        if type(originalNewIndex) == "function" then

            return originalNewIndex(target, key, value)

        end

        rawset(target, key, value)

    end

    local ok = pcall(setmetatable, equipmentState, nextMetatable)

    if ok then

        equipmentStateHooked = true

    end

    return ok

end

local function resolveEquippedEmotes()

    local currentData = DataController.CurrentData

    if not currentData then

        return nil, nil

    end

    if type(currentData.Get) == "function" then

        local ok, equipped = pcall(currentData.Get, currentData, "EquippedEmotes")

        if ok and type(equipped) == "table" then

            return currentData, equipped

        end

    end

    local data = type(currentData.Data) == "table" and currentData.Data or nil

    if not data then

        return currentData, nil

    end

    if type(data.EquippedEmotes) ~= "table" then

        data.EquippedEmotes = {}

    end

    return currentData, data.EquippedEmotes

end

local function setEquippedEmote(slotKey, emoteName, persist)

    local numericSlot = math.floor(tonumber(slotKey) or 0)

    if numericSlot < 1 or numericSlot > MAX_EMOTES then

        return false

    end

    if not isSupportedCosmetic(emoteName, "Emote") then

        return false

    end

    local currentData, equipped = resolveEquippedEmotes()

    if not currentData or type(equipped) ~= "table" then

        return false

    end

    local slotString = tostring(numericSlot)

    equipped[numericSlot] = nil

    equipped[slotString] = buildEmoteData(emoteName)

    State.emotes[slotString] = emoteName

    if type(currentData.Replicate) == "function" then

        pcall(currentData.Replicate, currentData, "EquippedEmotes")

    end

    if persist ~= false then

        saveConfig()

    end

    return true

end

local function applySavedEmotes()

    local applied = false

    for slotKey, emoteName in pairs(State.emotes) do

        if setEquippedEmote(slotKey, emoteName, false) then

            applied = true

        end

    end

    return applied

end

local function playLocalEmote(emoteName)

    return ok == true

end

            and not isNaturallyOwnedEmote(emoteName)

            and playLocalEmote(emoteName) then

            return

        end

        return originalUseEmoteByName(self, emoteName, ...)

    end

    emoteControllerHooked = true

    return true
